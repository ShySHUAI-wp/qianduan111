在`src/lerobot/scripts/lerobot_calibrate.py` 追加
```python
from lerobot.robots import (  # noqa: F401
    Robot,
    RobotConfig,
    hope_jr,
    koch_follower,
    lekiwi,
    make_robot_from_config,
    omx_follower,
    so100_follower,
    so101_follower,
    bi_so100_follower, # 新加
)
from lerobot.teleoperators import (  # noqa: F401
    Teleoperator,
    TeleoperatorConfig,
    homunculus,
    koch_leader,
    make_teleoperator_from_config,
    omx_leader,
    so100_leader,
    so101_leader,
    bi_so100_leader, # 新加
)
```
同时在`src/lerobot/scripts/lerobot_record.py` 追加
```python
if isinstance(teleop, list):
        teleop_keyboard = next((t for t in teleop if isinstance(t, KeyboardTeleop)), None)
        teleop_arm = next(
            (
                t
                for t in teleop
                if isinstance(
                    t,
                    (
                        so100_leader.SO100Leader
                        | so101_leader.SO101Leader
                        | koch_leader.KochLeader
                        | omx_leader.OmxLeader
                        | bi_so100_leader.BiSO100Leader # 新加
                    ),
                )
            ),
            None,
        )

        if not (teleop_arm and teleop_keyboard and len(teleop) == 2 and robot.name == "lekiwi_client"):
            raise ValueError(
                "For multi-teleop, the list must contain exactly one KeyboardTeleop and one arm teleoperator. Currently only supported for LeKiwi robot."
            )
```

本地遥操作
```bash
lerobot-teleoperate \
  --robot.type=bi_so100_follower \
  --robot.left_arm_port=/dev/cobot_follow_left \
  --robot.right_arm_port=/dev/cobot_follow_right \
  --robot.id=bimanual_follower \
  --teleop.type=bi_so100_leader \
  --teleop.left_arm_port=/dev/cobot_leader_left \
  --teleop.right_arm_port=/dev/cobot_leader_right \
  --teleop.id=bimanual_leader
```

```bash
lerobot-teleoperate \
    --robot.type=bi_so100_follower \
```
